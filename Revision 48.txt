#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n, m;
    cin >> n >> m;
    
    vector<int> a(n), b(m);
    for(auto &x : a) cin >> x;
    for(auto &y : b) cin >> y;
    
    if (n < m * 2) {
        cout << "NO\n";
        return;
    }
    
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    
    for (int i = 0; i < m; i++) {
        if (a[i] > b[i] || a[n - m + i] < b[i]) {
            cout << "NO\n";
            return;
        }
    }
    
    cout << "YES\n";
}

int main() {
    ios_base::sync_with_stdio(0); 
    cin.tie(0);
    
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
