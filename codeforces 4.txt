#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n;
    cin >> n;
    string s;
    cin >> s;

    int len = 1;
    for(int i = 1; i < n; i++) {
        if(s[i] != s[i-1]) len++;
    }

    int res = len;
    
    for(int i = 1; i < n - 1; i++) {
        int cur = len;
        
        cur -= (s[i-1] != s[i]);
        cur -= (s[i] != s[i+1]);
        cur += (s[i-1] != s[i+1]);
        
        res = min(res, cur);
    }

    cout << res << "\n";
}

int main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    int t;
    cin >> t;
    while(t--) solve();
    return 0;
}
