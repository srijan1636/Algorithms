#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for(int &x : a) cin >> x;
    for(int &x : b) cin >> x;
    
    int diff = 0, c1 = 0, h1 = 0, h0 = 0;
    
    for(int i = 0; i < n; i++) {
        if(a[i] != b[i]) {
            diff++;
            if(a[i] == 1) c1++;
        } else {
            if(a[i] == 1) h1 = 1;
            else h0 = 1;
        }
    }
    
    if(diff == 0) cout << 0 << "\n";
    else if(c1 % 2 != 0) cout << 1 << "\n";
    else if(c1 > 0) cout << 2 << "\n";
    else if(h1 && h0) cout << 2 << "\n";
    else cout << -1 << "\n";
}

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    
    int t;
    if(cin >> t) {
        while(t--) solve();
    }
    return 0;
}
