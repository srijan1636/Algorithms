#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n;
    long long k;
    cin >> n >> k;
    string s;
    cin >> s;
    
    int r = 0, b = 0;
    int m = n * 2;
    
    for (int i = 0; i < m; i++) {
        if (s[i] == '1') {
            int nxt = (i + 1 == m) ? 0 : i + 1;
            
            if (s[nxt] == '0') {
                (i % 2 == 0) ? r++ : b++;
            } else {
                (i % 2 == 0) ? b++ : r++;
            }
        }
    }
    
    cout << r << " " << b << "\n";
}

int main() {
    ios_base::sync_with_stdio(0); 
    cin.tie(0);
    
    int t;
    if (cin >> t) {
        while (t--) solve();
    }
    return 0;
}
