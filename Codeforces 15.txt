#include <bits/stdc++.h>
using namespace std;

#define pb push_back

void solve() {
    int n;
    cin >> n;
    
    vector<vector<int>> g(n + 1);
    for (int i = 2; i <= n; i++) {
        int p;
        cin >> p;
        g[p].pb(i);
    }
    
    int m;
    cin >> m;
    vector<int> target(n + 1, 0);
    for (int i = 0; i < m; i++) {
        int x;
        cin >> x;
        target[x] = 1;
    }
    
    vector<int> ans;
    
    auto dfs = [&](auto self, int u) -> int {
        int count = target[u];
        vector<int> active;
        
        for (int v : g[u]) {
            int sub = self(self, v);
            if (sub > 0) active.pb(v);
            count += sub;
        }
        
        if (!active.empty()) {
            if (target[u]) {
                for (int v : active) ans.pb(v);
            } else {
                for (int i = 1; i < (int)active.size(); i++) ans.pb(active[i]);
            }
        }
        return count;
    };
    
    dfs(dfs, 1);
    
    cout << ans.size() << "\n";
    for (int i = 0; i < (int)ans.size(); i++) {
        cout << ans[i] << (i + 1 == (int)ans.size() ? "" : " ");
    }
    cout << "\n";
}

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    
    int t;
    if (cin >> t) {
        while (t--) solve();
    }
    return 0;
}
