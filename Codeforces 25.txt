#include 
using namespace std;

void solve() {
    int n, k;
    cin >> n >> k;
    
    if (k < n || k > 2 * n - 1) {
        cout << -1 << "\n";
        return;
    }
    
    int same = 2 * n - k;
    int cur = same + 1;
    
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < n; j++) {
            if(i == j && i < same) {
                cout << i + 1 << " ";
            } else {
                cout << cur++ << " ";
            }
        }
        cout << "\n";
    }
}

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    
    int t;
    if(cin >> t) {
        while(t--) {
            solve();
        }
    }
    return 0;
}
