#include <bits/stdc++.h>
using namespace std;

int getmax(int arr[], int n){
    int max =INT_MIN;
    
    for(int i=0;i<n;i++){
        if(arr[i]>max){
            max=arr[i];
        }
    }
    return max;
}
int main(){
    int size;
    cin>>size;
    
    int arr[100];
    for(int i=0;i<size;i++){
        cin>>arr[i];
    }
    
    int maxval= getmax(arr,size);
    cout<<"Max Valus is: " <<maxval<<endl;
    return 0;
}
