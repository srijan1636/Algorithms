#include "bits/stdc++.h"
using namespace std;

void solve() {
    long long x0, y0, r;
    cin >> x0 >> y0 >> r;
    
    cout << x0 + r << " " << y0 << "\n";
}

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    
    int t;
    if (cin >> t) {
        while (t--) {
            solve();
        }
    }
    return 0;
}
