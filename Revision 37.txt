#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MOD = 998244353;
const int MAX = 1e6 + 5;

ll F[MAX], I[MAX];

ll pwr(ll a, ll b) {
    ll r = 1;
    a %= MOD;
    while (b > 0) {
        if (b & 1) r = (r * a) % MOD;
        a = (a * a) % MOD;
        b >>= 1;
    }
    return r;
}

void pre() {
    F[0] = I[0] = 1;
    for (int i = 1; i < MAX; i++) F[i] = (F[i - 1] * i) % MOD;
    I[MAX - 1] = pwr(F[MAX - 1], MOD - 2);
    for (int i = MAX - 2; i >= 1; i--) I[i] = (I[i + 1] * (i + 1)) % MOD;
}

ll C(int n, int r) {
    if (r < 0 || r > n) return 0;
    return F[n] * I[r] % MOD * I[n - r] % MOD;
}

void solve() {
    int n;
    string s;
    cin >> n >> s;

    int c[2] = {0, 0};
    int r[2] = {0, 0};

    for (int i = 0; i < n; i++) {
        int v = s[i] - '0';
        c[v]++;
        if (i == 0 || s[i] != s[i - 1]) r[v]++;
    }

    ll v1 = c[0] ? C(c[0] - 1, r[0] - 1) : 1;
    ll v2 = c[1] ? C(c[1] - 1, r[1] - 1) : 1;

    cout << (v1 * v2) % MOD << "\n";
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    pre();
    
    int tc;
    if (cin >> tc) {
        while (tc--) {
            solve();
        }
    }
    return 0;
}
