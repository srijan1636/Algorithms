#include "bits/stdc++.h"
using namespace std;

void solve() {
    int n;
    cin >> n;
    
    int f[105] = {0};
    for (int i = 0; i < n; i++) {
        int v;
        cin >> v;
        f[v]++;
    }
    
    int rem = n;
    while (rem > 0) {
        for (int i = 100; i >= 1; i--) {
            if (f[i] > 0) {
                cout << i << " ";
                f[i]--;
                rem--;
            }
        }
    }
    cout << "\n";
}

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    
    int t;
    if (cin >> t) {
        while (t--) {
            solve();
        }
    }
    return 0;
}
