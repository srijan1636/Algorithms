#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n;
    cin >> n;
    
    int z_cnt = 0, f = 0, l = 0;
    for(int i = 0, v; i < n; i++) {
        cin >> v;
        if(!i) f = v;
        if(i == n - 1) l = v;
        if(!v) z_cnt++;
    }
    
    cout << (z_cnt >= 2 ? f + l : -1) << "\n";
}

int main() {
    ios_base::sync_with_stdio(0); 
    cin.tie(0);
    
    int t; 
    if(cin >> t) {
        while(t--) solve();
    }
    return 0;
}
