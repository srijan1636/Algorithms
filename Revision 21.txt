#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n, q;
    cin >> n >> q;
    string s, t;
    cin >> s >> t;
    
    vector<int> pref_same(n + 1, 0);
    vector<int> pref_01(n + 1, 0);
    vector<int> pref_10(n + 1, 0);
    
    for (int i = 0; i < n; i++) {
        pref_same[i + 1] = pref_same[i] + (s[i] == t[i]);
        pref_01[i + 1] = pref_01[i] + (s[i] == '0' && t[i] == '1');
        pref_10[i + 1] = pref_10[i] + (s[i] == '1' && t[i] == '0');
    }
    
    while (q--) {
        int l, r;
        cin >> l >> r;
        
        int matches = pref_same[r] - pref_same[l - 1];
        int c01 = pref_01[r] - pref_01[l - 1];
        int c10 = pref_10[r] - pref_10[l - 1];
        
        if (matches >= abs(c01 - c10)) {
            cout << "YES\n";
        } else {
            cout << "NO\n";
        }
    }
}

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    
    int t;
    if (cin >> t) {
        while (t--) {
            solve();
        }
    }
    return 0;
}
