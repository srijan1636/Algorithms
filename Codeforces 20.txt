#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    int n;
    cin >> n;
    
    ll first = 0, last = 0, temp;
    for(int i = 0; i < n; i++) {
        cin >> temp;
        if(i == 0) first = temp;
        if(i == n - 1) last = temp;
    }
    
    cout << std::gcd(first, last) << "\n";
}

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    
    int t;
    if(cin >> t) {
        while(t--) {
            solve();
        }
    }
    return 0;
}
