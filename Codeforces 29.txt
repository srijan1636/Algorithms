#include "bits/stdc++.h"
using namespace std;

int a[1005];

int get_nxt(int x) {
    int s = 0;
    while(x > 0) {
        int d = x % 10;
        s += d * d;
        x /= 10;
    }
    return s;
}

void solve() {
    int n;
    cin >> n;
    
    for(int i = 0; i < n; i++) {
        cin >> a[i];
        for(int k = 0; k < 200; k++) {
            a[i] = get_nxt(a[i]);
        }
    }
    
    sort(a, a + n);
    
    long long ans = 0;
    long long cnt = 1;
    
    for(int i = 1; i < n; i++) {
        if(a[i] == a[i-1]) {
            cnt++;
        } else {
            ans += (cnt * (cnt - 1)) / 2;
            cnt = 1;
        }
    }
    ans += (cnt * (cnt - 1)) / 2;
    
    cout << ans << "\n";
}

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    
    int t;
    if(cin >> t) {
        while(t--) {
            solve();
        }
    }
    return 0;
}
