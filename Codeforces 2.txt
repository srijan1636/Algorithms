#include <bits/stdc++.h>
using namespace std;

#define ll long long

void solve() {
    int n;
    cin >> n;
    
    string a, b;
    cin >> a >> b;

    int count_a[2] = {0, 0};
    int count_b[2] = {0, 0};

    for (int i = 0; i < n; i++) {
        if (a[i] == '1') {
            count_a[i % 2]++;
        }
        if (b[i] == '1') {
            count_b[i % 2]++;
        }
    }

    if (count_a[0] == count_b[0] && count_a[1] == count_b[1]) {
        cout << "YES\n";
    } else {
        cout << "NO\n";
    }
}

int main() {
    ios_base::sync_with_stdio(false); 
    cin.tie(NULL);

    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    
    return 0;
}
