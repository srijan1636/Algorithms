#include <bits/stdc++.h>
using namespace std;

int main(){
    int n=6;
    if(n<=1){
        cout<<n<<" is not Prime number"<<endl;
        return 0;
    }
    
    bool isprime =true;
    
    for(int i=2;i<n;i++){
        if(n%i==0){
            isprime =false;
            break;
        }
    }
    
    if(isprime){
        cout<<n<<" Is A  PRIME NUMBER";
    } else {
        cout<<"It is Not prime number";
    }
    return 0;
}
