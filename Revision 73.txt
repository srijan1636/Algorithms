#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n;
    cin >> n;
    string s;
    cin >> s;

    int c[2] = {0, 0};
    int g[2] = {0, 0};

    for (int i = 0; i < n; i++) {
        int v = s[i] - '0';
        c[v]++;
        if (i == 0 || s[i] != s[i - 1]) {
            g[v]++;
        }
    }

    if (abs(c[0] - c[1]) > 2) {
        cout << "-1\n";
        return;
    }

    int ans = -1;
    
    for (int d = -1; d <= 1; d++) {
        if (abs(c[0] - c[1] - d) <= 1) {
            int t0 = 0, t1 = 0;
            
            if (d == 1) {
                t1 = min(g[0] - 1, g[1]);
                t0 = t1 + 1;
            } else if (d == -1) {
                t0 = min(g[0], g[1] - 1);
                t1 = t0 + 1;
            } else {
                t0 = min(g[0], g[1]);
                t1 = t0;
            }

            if (t0 >= 0 && t1 >= 0) {
                ans = max(ans, t0 + t1);
            }
        }
    }

    if (ans == -1) {
        cout << "-1\n";
    } else {
        cout << n - ans << "\n";
    }
}

int main() {
    ios_base::sync_with_stdio(0); 
    cin.tie(0);
    
    int t;
    if (cin >> t) {
        while (t--) {
            solve();
        }
    }
    return 0;
}
